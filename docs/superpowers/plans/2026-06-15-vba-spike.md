## 已核实的 windows 0.58 COM API（写 FFI 时以此为准）
- `windows::core::{IUnknown, IDispatch, VARIANT, BSTR, GUID, PCWSTR, Interface, w}`；`VARIANT: From<&str>/From<i32>/Default/Clone`；`i32::try_from(&VARIANT)`、`IUnknown::try_from(&VARIANT)`、`BSTR::try_from(&VARIANT)`；`IUnknown::cast::<IDispatch>() -> Result<IDispatch>`。
- `windows::Win32::System::Com::{CoInitializeEx, CoUninitialize, CLSIDFromProgID, COINIT_APARTMENTTHREADED, IDispatch, DISPPARAMS(Default), DISPATCH_FLAGS, DISPATCH_METHOD, DISPATCH_PROPERTYGET}`。
- `windows::Win32::System::Ole::GetActiveObject(rclsid: *const GUID, pvreserved: Option<*mut c_void>, ppunk: *mut Option<IUnknown>) -> Result<()>`。
- `IDispatch::GetIDsOfNames(riid:*const GUID, rgsznames:*const PCWSTR, cnames:u32, lcid:u32, rgdispid:*mut i32)`；`IDispatch::Invoke(dispidmember:i32, riid:*const GUID, lcid:u32, wflags:DISPATCH_FLAGS, pdispparams:*const DISPPARAMS, pvarresult:Option<*mut VARIANT>, pexcepinfo:Option<*mut EXCEPINFO>, puargerr:Option<*mut u32>)`。
- `DISPPARAMS { rgvarg:*mut VARIANT, rgdispidNamedArgs:*mut i32, cArgs:u32, cNamedArgs:u32 }`；`riid` 传 `GUID::zeroed()`（IID_NULL）；`lcid` 用 `0x0400`（LOCALE_USER_DEFAULT）。

```rust
//! PowerPoint COM 探测（VBA spike）。纯函数跨平台可测；COM 部分仅 Windows。

/// 从一段 VBA 模块代码里抽出可调用的过程名（Sub / Function）。
/// 忽略大小写与前导空白；忽略 Private；去重保序。
pub fn parse_macro_names(module_text: &str) -> Vec<String> {
    let mut out: Vec<String> = Vec::new();
    for line in module_text.lines() {
        let t = line.trim_start();
        let lower = t.to_ascii_lowercase();
        // 跳过 Private 过程（Application.Run 调不到）
        if lower.starts_with("private ") {
            continue;
        }
        let rest = lower.strip_prefix("public ").unwrap_or(&lower);
        let kw = if let Some(r) = rest.strip_prefix("sub ") {
            Some(r)
        } else {
            rest.strip_prefix("function ")
        };
        if let Some(r) = kw {
            // 名字 = 关键字后到 '(' 或空白前的标识符
            let name_lower_len = r.find(|c: char| c == '(' || c.is_whitespace()).unwrap_or(r.len());
            if name_lower_len > 0 {
                // 用原始大小写还原：在原 trimmed 行里取相同位置的名字
                let start = t.len() - rest.len() + (rest.len() - r.len());
                let name = &t[start..start + name_lower_len];
                if name.chars().all(|c| c.is_alphanumeric() || c == '_') && !out.iter().any(|n| n.eq_ignore_ascii_case(name)) {
                    out.push(name.to_string());
                }
            }
        }
    }
    out
}

#[cfg(test)]
mod tests {

    fn extracts_subs_and_functions_dedup_skip_private() {
        let code = "\

End Sub\n\
Public Sub NextQ\n\
End Sub\n\
Private Sub Helper()\n\
End Sub\n\

        assert_eq!(parse_macro_names(code), vec!["ShowAnswerA", "NextQ", "Score"]);

    #[test]
    fn empty_text_yields_nothing() {

#[cfg(windows)]
mod com {
    use windows::core::{w, Interface, BSTR, GUID, IUnknown, PCWSTR, VARIANT};
    use windows::Win32::System::Com::{
        CoInitializeEx, CoUninitialize, CLSIDFromProgID, DISPATCH_FLAGS, DISPATCH_METHOD,
        DISPATCH_PROPERTYGET, DISPPARAMS, IDispatch, COINIT_APARTMENTTHREADED,
    };
    use windows::Win32::System::Ole::GetActiveObject;

    const LOCALE_USER_DEFAULT: u32 = 0x0400;

    /// 进程级 COM 初始化守卫（每次命令调用一次）。
    pub struct ComGuard;
    impl ComGuard {
        pub fn init() -> Self {
            unsafe {
                let _ = CoInitializeEx(None, COINIT_APARTMENTTHREADED);
            }
            ComGuard
        }
    }
    impl Drop for ComGuard {
        fn drop(&mut self) {
            unsafe { CoUninitialize() };
        }
    }

    /// 连接已运行的 PowerPoint.Application，返回其 IDispatch。
    pub fn connect() -> Result<IDispatch, String> {
        unsafe {
            let clsid =
                CLSIDFromProgID(w!("PowerPoint.Application")).map_err(|e| format!("找不到 PowerPoint：{e}"))?;
            let mut unk: Option<IUnknown> = None;
            GetActiveObject(&clsid, None, &mut unk)
                .map_err(|_| "PowerPoint 未运行，请先打开 PowerPoint。".to_string())?;
            let unk = unk.ok_or_else(|| "GetActiveObject 返回空".to_string())?;
            unk.cast::<IDispatch>().map_err(|e| e.to_string())
        }
    }

    /// 调用 disp 的成员（method 或 propertyget），args 已是正序（内部反转）。
    fn invoke(disp: &IDispatch, name: &str, flags: DISPATCH_FLAGS, args: &[VARIANT]) -> Result<VARIANT, String> {
        unsafe {
            let wname: Vec<u16> = name.encode_utf16().chain(std::iter::once(0)).collect();
            let names = [PCWSTR(wname.as_ptr())];
            let mut dispid: i32 = 0;
            disp.GetIDsOfNames(&GUID::zeroed(), names.as_ptr(), 1, LOCALE_USER_DEFAULT, &mut dispid)
                .map_err(|e| format!("{name}: 找不到成员（{e}）"))?;
            // DISPPARAMS.rgvarg 是逆序
            let mut argv: Vec<VARIANT> = args.iter().rev().cloned().collect();
            let params = DISPPARAMS {
                rgvarg: if argv.is_empty() { std::ptr::null_mut() } else { argv.as_mut_ptr() },
                cArgs: argv.len() as u32,
                ..Default::default()
            };
            let mut result = VARIANT::default();
            disp.Invoke(dispid, &GUID::zeroed(), LOCALE_USER_DEFAULT, flags, &params, Some(&mut result), None, None)
                .map_err(|e| format!("{name}: 调用失败（{e}）"))?;
            Ok(result)
        }
    }

    /// 读对象属性（返回 IDispatch）。
    pub fn get_obj(disp: &IDispatch, name: &str) -> Result<IDispatch, String> {
        let v = invoke(disp, name, DISPATCH_PROPERTYGET, &[])?;
        IUnknown::try_from(&v)
            .map_err(|e| format!("{name} 不是对象（{e}）"))?
            .cast::<IDispatch>()
            .map_err(|e| e.to_string())
    }

    /// 读 i32 属性。
    pub fn get_i32(disp: &IDispatch, name: &str) -> Result<i32, String> {
        let v = invoke(disp, name, DISPATCH_PROPERTYGET, &[])?;
        i32::try_from(&v).map_err(|e| format!("{name} 不是整数（{e}）"))
    }

    /// 调方法返回 IDispatch（如 VBComponents.Item(i)）。
    pub fn call_obj(disp: &IDispatch, name: &str, args: &[VARIANT]) -> Result<IDispatch, String> {
        let v = invoke(disp, name, DISPATCH_METHOD | DISPATCH_PROPERTYGET, args)?;
        IUnknown::try_from(&v)
            .map_err(|e| format!("{name} 未返回对象（{e}）"))?
            .cast::<IDispatch>()
            .map_err(|e| e.to_string())
    }

    /// 调方法返回字符串（如 CodeModule.Lines(1, n)）。
    pub fn call_str(disp: &IDispatch, name: &str, args: &[VARIANT]) -> Result<String, String> {
        let v = invoke(disp, name, DISPATCH_METHOD | DISPATCH_PROPERTYGET, args)?;
        Ok(BSTR::try_from(&v).map_err(|e| format!("{name} 未返回字符串（{e}）"))?.to_string())
    }

    /// 调方法忽略返回（如 Application.Run）。
    pub fn call_void(disp: &IDispatch, name: &str, args: &[VARIANT]) -> Result<(), String> {
        invoke(disp, name, DISPATCH_METHOD, args).map(|_| ())
    }
}
```

/// 列出当前活动演示文稿里可调用的 VBA 宏名。
#[cfg(windows)]
pub fn list_macros() -> Result<Vec<String>, String> {
    let _co = com::ComGuard::init();
    let app = com::connect()?;
    let pres = com::get_obj(&app, "ActivePresentation").map_err(|_| "没有活动演示文稿（请打开一个 .pptm）".to_string())?;
    let vbproj = com::get_obj(&pres, "VBProject")
        .map_err(|_| "无法访问 VBProject：请在 PowerPoint 信任中心开启‘信任对 VBA 工程对象模型的访问’。".to_string())?;
    let comps = com::get_obj(&vbproj, "VBComponents")?;
    let count = com::get_i32(&comps, "Count")?;
    let mut names: Vec<String> = Vec::new();
    for i in 1..=count {
        let comp = com::call_obj(&comps, "Item", &[windows::core::VARIANT::from(i)])?;
        let code = com::get_obj(&comp, "CodeModule")?;
        let lines = com::get_i32(&code, "CountOfLines")?;
        if lines > 0 {
            let text = com::call_str(
                &code,
                "Lines",
                &[windows::core::VARIANT::from(1i32), windows::core::VARIANT::from(lines)],
            )?;
            for n in parse_macro_names(&text) {
                if !names.iter().any(|x| x == &n) {
                    names.push(n);
                }
            }
        }
    }
    Ok(names)
}

#[cfg(not(windows))]
pub fn list_macros() -> Result<Vec<String>, String> {
    Err("仅 Windows 支持 PowerPoint COM".to_string())
}

// 临时 VBA spike 面板：探测当前 PPT 的 VBA 宏，点一个触发（导播本机 IPC）。
import { ref } from "vue";
import { api } from "../utils/api";

const macros = ref<string[]>([]);
const error = ref<string | null>(null);
const loading = ref(false);

async function detect() {
  loading.value = true;
  error.value = null;
  try {
    macros.value = await api.vbaListMacros();
  } catch (e) {
    error.value = String(e);
    macros.value = [];
  } finally {
    loading.value = false;
  }
}

async function run(name: string) {
  error.value = null;
  try {
    await api.vbaRunMacro(name);
  } catch (e) {
    error.value = String(e);
  }
}

    <p v-if="error" class="error">{{ error }}</p>
    <ul v-if="macros.length" class="macros">
      <li v-for="m in macros" :key="m">
